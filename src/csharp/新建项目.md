### Models/Result/ApiResult.cs

```
namespace MiniApiDemo.Models.Result;

public class ApiResult<T>
{
    public int Code { get; set; }
    public string Msg { get; set; } = string.Empty;
    public T? Data { get; set; }

    public static ApiResult<T> Success(T? data, string msg = "ok")
    {
        return new ApiResult<T> { Code = 200, Msg = msg, Data = data };
    }
    public static ApiResult<T> Fail(string msg, int code = 500)
    {
        return new ApiResult<T> { Code = code, Msg = msg };
    }
}
```

## 6、Common

### Common/Filters/GlobalExceptionFilter.cs

```
using Microsoft.AspNetCore.Mvc;
using Microsoft.AspNetCore.Mvc.Filters;
using MiniApiDemo.Models.Result;

namespace MiniApiDemo.Common.Filters;

public class GlobalExceptionFilter : IExceptionFilter
{
    public void OnException(ExceptionContext context)
    {
        context.Result = new ObjectResult(ApiResult<object>.Fail(context.Exception.Message))
        {
            StatusCode = 500
        };
        context.ExceptionHandled = true;
    }
}
```

### Common/Helpers/RedisHelper.cs（单例复用 ConnectionMultiplexer）

```
using StackExchange.Redis;

namespace MiniApiDemo.Common.Helpers;

public class RedisHelper
{
    private readonly IDatabase _db;
    public RedisHelper(IConnectionMultiplexer muxer)
    {
        _db = muxer.GetDatabase();
    }

    public async Task<bool> SetAsync(string key, string value, TimeSpan? expire = null)
    {
        return await _db.StringSetAsync(key, value, expire);
    }
    public async Task<string?> GetAsync(string key)
    {
        return await _db.StringGetAsync(key);
    }
    public async Task<bool> DelAsync(string key)
    {
        return await _db.KeyDeleteAsync(key);
    }
}
```

### Common/Extensions/ServiceCollectionExtensions.cs（集中注册）

```
using Microsoft.EntityFrameworkCore;
using MiniApiDemo.Common.Filters;
using MiniApiDemo.Common.Helpers;
using MiniApiDemo.Data;
using MiniApiDemo.Services.Impl;
using MiniApiDemo.Services.Interfaces;
using StackExchange.Redis;

namespace MiniApiDemo.Common.Extensions;

public static class ServiceCollectionExtensions
{
    public static void AddBusinessServices(this IServiceCollection services, IConfiguration cfg)
    {
        // DbContext MySql
        services.AddDbContext<AppDbContext>(opt =>
        {
            opt.UseMySQL(cfg.GetConnectionString("MySql")!);
        });

        // Redis
        var redisConn = cfg.GetConnectionString("Redis")!;
        var muxer = ConnectionMultiplexer.Connect(redisConn);
        services.AddSingleton<IConnectionMultiplexer>(muxer);
        services.AddScoped<RedisHelper>();

        // Service
        services.AddScoped<IUserService, UserService>();

        // 全局过滤器
        services.AddMvc(o =>
        {
            o.Filters.Add<GlobalExceptionFilter>();
        });
    }
}
```

### Services/Impl/UserService.cs（演示 Redis 缓存）

```
using MiniApiDemo.Common.Helpers;
using MiniApiDemo.Data;
using MiniApiDemo.Models.Dto;
using MiniApiDemo.Models.Entity;
using MiniApiDemo.Services.Interfaces;
using Microsoft.EntityFrameworkCore;

namespace MiniApiDemo.Services.Impl;

public class UserService : IUserService
{
    private readonly AppDbContext _db;
    private readonly RedisHelper _redis;

    public UserService(AppDbContext db, RedisHelper redis)
    {
        _db = db;
        _redis = redis;
    }

    public async Task<UserDto?> GetByIdAsync(int id)
    {
        var cacheKey = $"user:{id}";
        var cacheVal = await _redis.GetAsync(cacheKey);
        if (!string.IsNullOrEmpty(cacheVal))
        {
            return System.Text.Json.JsonSerializer.Deserialize<UserDto>(cacheVal);
        }

        var entity = await _db.Users.FindAsync(id);
        if (entity == null) return null;

        var dto = new UserDto
        {
            Id = entity.Id,
            UserName = entity.UserName,
            Phone = entity.Phone,
            CreateTime = entity.CreateTime
        };
        // 写入缓存，5分钟过期
        var json = System.Text.Json.JsonSerializer.Serialize(dto);
        await _redis.SetAsync(cacheKey, json, TimeSpan.FromMinutes(5));
        return dto;
    }

    public async Task AddAsync(UserAddDto dto)
    {
        var user = new User
        {
            UserName = dto.UserName,
            Phone = dto.Phone
        };
        _db.Users.Add(user);
        await _db.SaveChangesAsync();
    }
}
```

## 8、Controllers

### Controllers/BaseController.cs

```
using Microsoft.AspNetCore.Mvc;
using MiniApiDemo.Models.Result;

namespace MiniApiDemo.Controllers;

[ApiController]
[Route("api/[controller]")]
public class BaseController : ControllerBase
{
    protected IActionResult OkData<T>(T data)
    {
        return Ok(ApiResult<T>.Success(data));
    }
    protected IActionResult FailMsg(string msg)
    {
        return Ok(ApiResult<object>.Fail(msg));
    }
}
```

### Controllers/UserController.cs

```
using Microsoft.AspNetCore.Mvc;
using MiniApiDemo.Models.Dto;
using MiniApiDemo.Services.Interfaces;

namespace MiniApiDemo.Controllers;

public class UserController : BaseController
{
    private readonly IUserService _userService;
    public UserController(IUserService userService)
    {
        _userService = userService;
    }

    [HttpGet("{id:int}")]
    public async Task<IActionResult> Get(int id)
    {
        var data = await _userService.GetByIdAsync(id);
        if (data == null) return FailMsg("用户不存在");
        return OkData(data);
    }

    [HttpPost]
    public async Task<IActionResult> Add(UserAddDto dto)
    {
        await _userService.AddAsync(dto);
        return OkData("新增成功");
    }
}
```

## 9、Program.cs（net11 顶级语句，内置 Swagger）

```
using MiniApiDemo.Common.Extensions;

var builder = WebApplication.CreateBuilder(args);

builder.Services.AddControllers();
builder.Services.AddEndpointsApiExplorer();
builder.Services.AddSwaggerGen(c =>
{
    c.SwaggerDoc("v1", new Microsoft.OpenApi.Models.OpenApiInfo
    {
        Title = "MiniApiDemo接口文档",
        Version = "v1"
    });
});

// 注册业务、mysql、redis、过滤器
builder.Services.AddBusinessServices(builder.Configuration);

var app = builder.Build();

if (app.Environment.IsDevelopment())
{
    app.UseSwagger();
    app.UseSwaggerUI();
}

app.UseHttpsRedirection();
app.UseAuthorization();
app.MapControllers();

app.Run();
```
